# --- Busqueda vectorial de cursos --------------------------------------------
# Indice semantico sobre el catalogo de +8.000 cursos (GET /cursos/buscar).
#
# Ambas faltaban en este archivo aunque el codigo ya las importaba: la funcion
# de busqueda solo corria en la maquina donde se genero el indice, y en una
# instalacion limpia (CI, Docker, OCI) el import fallaba.
#
# `sentence-transformers` arrastra PyTorch (~2 GB). Por eso `busqueda/almacen.py`
# lo importa de forma diferida: si la dependencia o el indice no estan, la API
# arranca igual y `/cursos/buscar` responde con lista vacia en vez de un 500.
chromadb==1.5.9
sentence-transformers==3.3.1

# --- Usuarios y autenticacion (Semana 5) --------------------------------------
# SQLAlchemy sirve a SQLite (desarrollo/pruebas, cero infraestructura) y a
# Postgres (produccion en OCI, ver docker-compose.yml) con el mismo codigo:
# el dialecto lo decide ATHENIA_DB_URL, no el runtime.
sqlalchemy==2.0.36
psycopg2-binary==2.9.10
# Version fijada a proposito: passlib 1.7.4 con bcrypt>=4.1 choca con un
# atributo interno (`__about__`) que passlib espera y bcrypt dejo de exponer,
# y lanza un warning en cada hash. bcrypt 4.0.1 es la ultima version sin ese
# problema.
passlib[bcrypt]==1.7.4
bcrypt==4.0.1
python-jose[cryptography]==3.3.0

# --- Asistente conversacional (OpenAI) ---------------------------------------
# `asistente/motor_openai.py` importa este paquete de forma diferida, igual

# sin ATHENIA_OPENAI_API_KEY), la API arranca igual y el Asistente responde
# solo con los cursos encontrados, sin redaccion, en vez de un 500.
openai==1.59.6
